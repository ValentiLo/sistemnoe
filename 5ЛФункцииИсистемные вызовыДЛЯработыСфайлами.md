# Полный набор: 25 примеров по теме «Структуры хранения информации о файле. Функции и системные вызовы для работы с файлами»

Ниже — **25 примеров**: 20 на функции стандартной библиотеки (`FILE *`) + 5 на системные вызовы (`int fd`).

---

## 📖 Часть 1. Функции стандартной библиотеки (примеры 1–20)

### Пример 1. Чтение файла посимвольно (`fgetc`)

```c
#include <stdio.h>

int main(void) {
    FILE *fp = fopen("hello.txt", "r");
    if (!fp) return 1;

    int c;
    while ((c = fgetc(fp)) != EOF)
        printf("%c", c);

    fclose(fp);
    return 0;
}
```

**Что делает:** читает файл **по одному символу** до конца (`EOF`).

---

### Пример 2. Чтение файла построчно (`fgets`)

```c
#include <stdio.h>

int main(void) {
    FILE *fp = fopen("hello.txt", "r");
    if (!fp) return 1;

    char buf[256];
    while (fgets(buf, sizeof(buf), fp))
        printf("%s", buf);

    fclose(fp);
    return 0;
}
```

**Что делает:** читает файл **по строкам** (до 255 символов за раз).

---

### Пример 3. Чтение с проверкой на ошибку (`perror`)

```c
#include <stdio.h>

int main(void) {
    FILE *fp = fopen("hello.txt", "r");
    if (fp == NULL) {
        perror("fopen");
        return 1;
    }
    fclose(fp);
    return 0;
}
```

**Что делает:** если файл не открылся — выводит **сообщение об ошибке** через `perror`.

### Пример 4. Подсчёт символов

```c
#include <stdio.h>

int main(void) {
    FILE *fp = fopen("hello.txt", "r");
    if (!fp) return 1;

    int c, count = 0;
    while ((c = fgetc(fp)) != EOF) count++;

    printf("Символов: %d\n", count);
    fclose(fp);
    return 0;
}
```

**Что делает:** считает **количество символов** в файле.

### Пример 5. Подсчёт строк

```c
#include <stdio.h>

int main(void) {
    FILE *fp = fopen("hello.txt", "r");
    if (!fp) return 1;

    int c, lines = 0;
    while ((c = fgetc(fp)) != EOF)
        if (c == '\n') lines++;

    printf("Строк: %d\n", lines);
    fclose(fp);
    return 0;
}
```

**Что делает:** считает **количество строк** (по символу `\n`).

### Пример 7. Запись посимвольно (`fputc`)

```c

int main(void) {
    FILE *fp = fopen("out.txt", "w");
```

**Что делает:** записывает каждый символ отдельно через `fputc`.

### Пример 11. Копирование файла

```c
#include <stdio.h>

int main(void) {
    FILE *in = fopen("in.txt", "r");
    FILE *out = fopen("out.txt", "w");
    if (!in || !out) return 1;

    int c;
    while ((c = fgetc(in)) != EOF)
        fputc(c, out);

    fclose(in);
    fclose(out);
    return 0;
}
```

**Что делает:** **копирует** содержимое одного файла в другой.

### Пример 14. Поиск слова в файле

```c

int main(void) {

    if (!fp) return 1;
```

**Что делает:** ищет **слово** в файле и выводит номер строки.

### Пример 15. Слияние двух файлов

```c
#include <stdio.h>

int main(void) {
    FILE *f1 = fopen("a.txt", "r");
    FILE *f2 = fopen("b.txt", "r");
    FILE *out = fopen("c.txt", "w");
    if (!f1 || !f2 || !out) return 1;

    int c;
    while ((c = fgetc(f1)) != EOF) fputc(c, out);
    while ((c = fgetc(f2)) != EOF) fputc(c, out);

    fclose(f1);
    fclose(f2);
    fclose(out);
    return 0;
}
```

**Что делает:** **сливает** два файла в один.

### Пример 16. Размер файла через `fseek` + `ftell`

```c
#include <stdio.h>

int main(void) {
    FILE *fp = fopen("hello.txt", "r");
    if (!fp) return 1;

    fseek(fp, 0, SEEK_END);
    long size = ftell(fp);
    printf("Размер: %ld байт\n", size);

    fclose(fp);
    return 0;
}
```

**Что делает:** **перемещает** указатель в конец и получает **размер** файла.

### Пример 18. Перемотка в начало (`rewind`)

```c
#include <stdio.h>

int main(void) {
    FILE *fp = fopen("hello.txt", "r");
    if (!fp) return 1;

    while ((c = fgetc(fp)) != EOF) putchar(c);
```

**Что делает:** читает файл **дважды** — использует `rewind`.

### Пример 19. Метаданные через `stat`

```c
#include <stdio.h>
#include <sys/stat.h>

int main(void) {
    struct stat st;
    if (stat("hello.txt", &st) == -1) {
        perror("stat");
        return 1;
    }

    printf("Размер: %ld байт\n", (long)st.st_size);
    printf("Тип: %s\n",
           S_ISREG(st.st_mode) ? "файл" :
           S_ISDIR(st.st_mode) ? "каталог" : "другое");

    return 0;
}
```

**Что делает:** получает **метаданные** файла через `stat`.

## ⚙️ Часть 2. Системные вызовы (примеры 21–25)

### Пример 21. Открытие файла через `open`

```c
#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>

int main(void) {
    int fd = open("hello.txt", O_RDONLY);
    if (fd == -1) {
        perror("open");
        return 1;
    }
    printf("Дескриптор: %d\n", fd);
    close(fd);
    return 0;
}
```

**Что делает:** открывает файл **системным вызовом** `open`. Возвращает **файловый дескриптор** (`int`).

**Разбор флагов:**
- `O_RDONLY` — только чтение.
- `O_WRONLY` — только запись.
- `O_RDWR` — чтение и запись.
- `O_CREAT` — создать, если нет.
- `O_TRUNC` — обрезать до нуля.

---

### Пример 22. Чтение через `read`

```c
#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>

int main(void) {
    int fd = open("hello.txt", O_RDONLY);
    if (fd == -1) return 1;

    char buf[256];
    ssize_t n;
    while ((n = read(fd, buf, sizeof(buf))) > 0)
        write(1, buf, n);   // вывод в stdout

    close(fd);
    return 0;
}
```

**Что делает:** читает файл **блоками** через системный вызов `read`.

**Разбор:**
- `read(fd, buf, count)` — читает до `count` байт.
- Возвращает число прочитанных байт, `0` — конец, `-1` — ошибка.
- `write(1, buf, n)` — вывод в **stdout** (дескриптор 1).

---

### Пример 23. Запись через `write`

```c
#include <fcntl.h>
#include <unistd.h>

int main(void) {
    int fd = open("out.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd == -1) return 1;

    const char *msg = "Hello\n";
    write(fd, msg, 6);

    close(fd);
    return 0;
}
```

**Что делает:** записывает строку в файл через `write`.

**Разбор:**
- `O_WRONLY | O_CREAT | O_TRUNC` — открыть для записи, создать, обрезать.
- `0644` — права доступа: владелец — чтение/запись, остальные — чтение.
- `write(fd, buf, count)` — записывает до `count` байт.

---

### Пример 24. Позиционирование через `lseek`

```c
#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>

int main(void) {
    int fd = open("hello.txt", O_RDONLY);
    if (fd == -1) return 1;

    lseek(fd, 0, SEEK_END);
    off_t size = lseek(fd, 0, SEEK_CUR);
    printf("Размер: %ld байт\n", (long)size);

    close(fd);
    return 0;
}
```

**Что делает:** перемещает позицию через `lseek` и получает **размер файла**.

**Разбор:**
- `lseek(fd, offset, origin)` — перемещает указатель.
- `SEEK_SET` — от начала, `SEEK_CUR` — от текущей, `SEEK_END` — от конца.
- Возвращает **новую позицию**.

---

### Пример 25. Метаданные через `fstat`

```c
#include <fcntl.h>
#include <sys/stat.h>
#include <unistd.h>
#include <stdio.h>

int main(void) {
    int fd = open("hello.txt", O_RDONLY);
    if (fd == -1) return 1;

    struct stat st;
    if (fstat(fd, &st) == -1) return 1;

    printf("Размер: %ld\n", (long)st.st_size);
    printf("Inode: %ld\n", (long)st.st_ino);
    printf("Тип: %s\n",
           S_ISREG(st.st_mode) ? "файл" :
           S_ISDIR(st.st_mode) ? "каталог" : "другое");

    close(fd);
    return 0;
}
```

**Что делает:** получает метаданные по **открытому дескриптору** через `fstat`.

**Отличие от `stat`:**
- `stat(path, &st)` — по **имени файла**.
- `fstat(fd, &st)` — по **открытому дескриптору**.
- `lstat(path, &st)` — не разыменовывает **символические ссылки**.

| Задача | Функция (`FILE *`) | Системный вызов (`int fd`) |
|---|---|---|
| Открыть | `fopen` | `open` |
| Закрыть | `fclose` | `close` |
| Читать символ | `fgetc` | `read` (по 1 байту) |

| Писать символ | `fputc` | `write` (по 1 байту) |
| Писать блок | `fwrite` | `write` |
| Писать строку | `fputs` | — |
| Форматировать | `fprintf` | — |
| Позиция | `fseek`, `ftell` | `lseek` |

| Метаданные | `stat`, `fstat` | `stat`, `fstat` |
| Буферизация | Есть | Нет |

## 📌 Режимы открытия — сравнение

### `fopen`

| Режим | Значение |
|---|---|
| `"r"` | Чтение |
| `"w"` | Запись (перезапись) |
| `"a"` | Дозапись |
| `"r+"` | Чтение и запись |
| `"w+"` | Чтение и запись (перезапись) |
| `"a+"` | Чтение и дозапись |
| `"rb"`, `"wb"` | Бинарные режимы |

### `open`

| Флаг | Значение |
|---|---|
| `O_RDONLY` | Только чтение |
| `O_WRONLY` | Только запись |
| `O_RDWR` | Чтение и запись |
| `O_CREAT` | Создать, если нет |
| `O_TRUNC` | Обрезать до нуля |
| `O_APPEND` | Писать в конец |
| `O_EXCL` | Ошибка, если файл есть |

## 🎯 Что важно запомнить

### Структуры хранения информации о файле

1. **`FILE *`** — высокоуровневая структура (буфер, дескриптор, позиция, флаги).
2. **`int fd`** — файловый дескриптор (низкоуровневый).
3. **`struct stat`** — метаданные файла (размер, тип, права, время).

### Функции

- Работают с `FILE *`.
- **Буферизованы** — быстрее при мелких операциях.
- **Платформо-независимы** — стандарт C.

### Системные вызовы

- Работают с `int fd`.
- **Не буферизованы** — каждое обращение к диску.
- **POSIX** — Unix/Linux.
- Прямой доступ к ядру ОС.
