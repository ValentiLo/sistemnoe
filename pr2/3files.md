# Задача: заменить содержимое самого длинного файла на содержимое самого короткого

**Условие:** даны три файла разного размера. Имена вводит пользователь. Нужно заменить содержимое **самого длинного** файла на содержимое **самого короткого**.

---

## 🔍 Алгоритм

1. Запросить у пользователя **три имени файлов**.
2. **Определить размер** каждого файла.
3. Найти **самый длинный** (max) и **самый короткий** (min).
4. **Прочитать** содержимое самого короткого.
5. **Записать** это содержимое в самый длинный файл (режим `"w"` — перезапишет содержимое).
6. Закрыть файлы.

---

## 💻 Решение на C

```c
#include <stdio.h>
#include <stdlib.h>

/* Возвращает размер файла в байтах, или -1 при ошибке */
long file_size(const char *filename)
{
    FILE *f = fopen(filename, "rb");   // rb — бинарный режим для точного размера
    if (f == NULL) return -1;

    fseek(f, 0, SEEK_END);   // переходим в конец
    long size = ftell(f);    // получаем позицию = размер
    fclose(f);

    return size;
}

/* Копирует содержимое файла src в файл dst */
int copy_file(const char *src, const char *dst)
{
    FILE *in = fopen(src, "rb");
    if (in == NULL) {
        perror("Открытие источника");
        return -1;
    }

    FILE *out = fopen(dst, "wb");   // "w" перезапишет содержимое
    if (out == NULL) {
        perror("Открытие приёмника");
        fclose(in);
        return -1;
    }

    int c;
    while ((c = fgetc(in)) != EOF) {
        fputc(c, out);
    }

    fclose(in);
    fclose(out);
    return 0;
}

int main(void)
{
    char names[3][256];   // три имени файлов

    /* 1. Ввод имён файлов */
    printf("Введите имена трёх файлов:\n");
    for (int i = 0; i < 3; i++) {
        printf("Файл %d: ", i + 1);
        scanf("%255s", names[i]);
    }

    /* 2. Определяем размеры */
    long sizes[3];
    for (int i = 0; i < 3; i++) {
        sizes[i] = file_size(names[i]);
        if (sizes[i] < 0) {
            fprintf(stderr, "Ошибка: не удалось открыть '%s'\n", names[i]);
            return 1;
        }
    }

    /* 3. Находим индексы самого длинного и самого короткого */
    int idx_max = 0;
    int idx_min = 0;
    for (int i = 1; i < 3; i++) {
        if (sizes[i] > sizes[idx_max]) idx_max = i;
        if (sizes[i] < sizes[idx_min]) idx_min = i;
    }

    printf("\nСамый длинный: %s (%ld байт)\n", names[idx_max], sizes[idx_max]);
    printf("Самый короткий: %s (%ld байт)\n", names[idx_min], sizes[idx_min]);

    /* 4. Копируем содержимое короткого в длинный */
    if (copy_file(names[idx_min], names[idx_max]) != 0) {
        fprintf(stderr, "Ошибка копирования\n");
        return 1;
    }

    printf("\nГотово: содержимое '%s' записано в '%s'\n",
           names[idx_min], names[idx_max]);

    return 0;
}
```

**Создадим три файла:**

cat long.txt

Содержимое `long.txt` заменилось на содержимое `short.txt`. ✅

## 🔍 Разбор ключевых моментов

### 1. Функция `file_size`

```c
fseek(f, 0, SEEK_END);   // переходим в конец файла
long size = ftell(f);    // ftell возвращает текущую позицию = размер
```

- **`fseek(f, 0, SEEK_END)`** — перемещает указатель в **конец** файла.
- **`ftell(f)`** — возвращает **текущую позицию** — это и есть **размер файла** в байтах.

### 2. Поиск максимума и минимума

```c
int idx_max = 0, idx_min = 0;
for (int i = 1; i < 3; i++) {
    if (sizes[i] > sizes[idx_max]) idx_max = i;
    if (sizes[i] < sizes[idx_min]) idx_min = i;
}
```

- Храним **индексы** (0, 1, 2), а не сами значения.
- Так мы знаем, **какое имя файла** соответствует максимуму/минимуму.

### 3. Режим `"wb"` — перезапись

```c
FILE *out = fopen(dst, "wb");
```

- `"wb"` — запись в **бинарном** режиме.
- **Содержимое файла будет полностью заменено.**
- Если файл был длиннее — он **усечётся** до новой длины.

### 4. Копирование посимвольно

```c
int c;
while ((c = fgetc(in)) != EOF) {
    fputc(c, out);
}
```

- **`fgetc`** — читает один символ.
- **`fputc`** — записывает один символ.
- Работает для **любых** файлов — текстовых и бинарных.

### 5. Почему `int c`, а не `char c`?

`fgetc` возвращает **`int`**, чтобы можно было отличить обычный символ от `EOF` (`-1`). Если использовать `char`, на некоторых системах `EOF` превратится в `255` — цикл не завершится.

## ⚠️ Возможные ошибки

| Ошибка | Причина | Решение |
|--------|---------|---------|
| `fopen` вернул `NULL` | Файл не существует | Проверить имена |
| Бесконечный цикл копирования | Использован `char c` вместо `int c` | Заменить на `int` |
| Файл не изменился | Не вызван `fclose` | Всегда закрывать файлы |
| Программа падает | Нет проверки на `NULL` | Добавить `if (f == NULL)` |

## 🎯 Упрощённый вариант (без функций)

Если нужно **короче** — можно без вспомогательных функций:

```c
#include <stdio.h>

int main(void)
{
    char names[3][256];
    long sizes[3];
    FILE *f;

    /* Ввод имён и определение размеров */
    for (int i = 0; i < 3; i++) {
        scanf("%255s", names[i]);
        f = fopen(names[i], "rb");
        if (f == NULL) { perror(names[i]); return 1; }
        fseek(f, 0, SEEK_END);
        sizes[i] = ftell(f);
        fclose(f);
    }

    /* Поиск max и min */
    int imax = 0, imin = 0;
    for (int i = 1; i < 3; i++) {
        if (sizes[i] > sizes[imax]) imax = i;
        if (sizes[i] < sizes[imin]) imin = i;
    }

    /* Копирование */
    FILE *in = fopen(names[imin], "rb");
    FILE *out = fopen(names[imax], "wb");
    if (in == NULL || out == NULL) { perror("fopen"); return 1; }

    int c;
    while ((c = fgetc(in)) != EOF)
        fputc(c, out);

    fclose(in);
    fclose(out);
    return 0;
}
```
